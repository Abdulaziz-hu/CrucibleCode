cmake_minimum_required(VERSION 3.21)

project(CrucibleCode
    VERSION 0.9.0
    DESCRIPTION "CrucibleCode - a lightweight cross-platform IDE for C, C++ and C# built on Qt 6"
    HOMEPAGE_URL "https://github.com/Abdulaziz-hu/CrucibleCode"
    LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(CRUCIBLE_USE_KSYNTAX
       "Use KSyntaxHighlighting (MIT) for syntax highlighting (300+ languages)" ON)
option(CRUCIBLE_BUNDLE_KSYNTAX
       "Build KSyntaxHighlighting from source into CrucibleCode when it is not installed (needs git + network once)" ON)
set(CRUCIBLE_KF_TAG "v6.5.0" CACHE STRING "KDE Frameworks tag used when bundling KSyntaxHighlighting")

find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network)
find_package(Qt6 6.4 QUIET OPTIONAL_COMPONENTS Svg Pdf PdfWidgets)
qt_standard_project_setup()

set(CRUCIBLE_HAVE_KSYNTAX OFF)
if(CRUCIBLE_USE_KSYNTAX)
    find_package(KF6SyntaxHighlighting QUIET)                    # an installed copy wins
    if(NOT KF6SyntaxHighlighting_FOUND AND CRUCIBLE_BUNDLE_KSYNTAX)
        include(cmake/BundleKSyntax.cmake)
        crucible_build_ksyntax(CRUCIBLE_KSYNTAX_PREFIX)
        if(CRUCIBLE_KSYNTAX_PREFIX)
            find_package(KF6SyntaxHighlighting CONFIG QUIET PATHS "${CRUCIBLE_KSYNTAX_PREFIX}" NO_DEFAULT_PATH)
        endif()
    endif()
    if(KF6SyntaxHighlighting_FOUND)
        set(CRUCIBLE_HAVE_KSYNTAX ON)
    endif()
endif()
message(STATUS "KSyntaxHighlighting: ${CRUCIBLE_HAVE_KSYNTAX}")
message(STATUS "Qt SVG: ${Qt6Svg_FOUND}   Qt PDF viewer: ${Qt6PdfWidgets_FOUND}")

qt_add_executable(cruciblecode WIN32 MACOSX_BUNDLE
    src/main.cpp
    src/MainWindow.h        src/MainWindow.cpp
    src/CodeEditor.h        src/CodeEditor.cpp
    src/SyntaxHighlighter.h src/SyntaxHighlighter.cpp
    src/ThemeManager.h      src/ThemeManager.cpp
    src/BuildManager.h      src/BuildManager.cpp
    src/FindBar.h           src/FindBar.cpp
    src/SearchPanel.h       src/SearchPanel.cpp
    src/SettingsDialog.h    src/SettingsDialog.cpp
    src/PtyProcess.h        src/PtyProcess.cpp
    src/TerminalScreen.h    src/TerminalScreen.cpp
    src/TerminalView.h      src/TerminalView.cpp
    src/TerminalPanel.h     src/TerminalPanel.cpp
    src/QuickOpenDialog.h   src/QuickOpenDialog.cpp
    src/FileViewers.h       src/FileViewers.cpp
    src/SpellChecker.h      src/SpellChecker.cpp
    src/RunHelper.h         src/RunHelper.cpp
    src/AiCompletion.h      src/AiCompletion.cpp
    src/AiSetupDialog.h     src/AiSetupDialog.cpp
    src/LiveServer.h        src/LiveServer.cpp
    src/GitManager.h        src/GitManager.cpp
    src/GitPanel.h          src/GitPanel.cpp
    src/Icons.h
    src/GitExplorerDelegate.h   src/GitExplorerDelegate.cpp
    src/LspManager.h            src/LspManager.cpp
    src/LspTypes.h
    src/LspClient.h             src/LspClient.cpp
    src/Updater.h               src/Updater.cpp
    src/UpdateDialog.h          src/UpdateDialog.cpp
    src/SnippetDialog.h         src/SnippetDialog.cpp
    src/PerfMonitor.h           src/PerfMonitor.cpp
    src/Snippets.h              src/Snippets.cpp
    src/CodeEditorSnippets.cpp
    src/ShortcutManager.h       src/ShortcutManager.cpp
    src/ThemeEditorDialog.h     src/ThemeEditorDialog.cpp
    src/ThemeCss.h              src/ThemeCss.cpp
    src/ThemeLibrary.h          src/ThemeLibrary.cpp
    src/ThemeMarketplace.h      src/ThemeMarketplace.cpp
    src/ThemeMarketplaceDialog.h src/ThemeMarketplaceDialog.cpp
    src/ProjectTemplates.h      src/ProjectTemplates.cpp
    src/TemplateDialogs.h       src/TemplateDialogs.cpp
    src/AppInfo.h
    src/MemTrim.h
)

target_link_libraries(cruciblecode PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
if(Qt6Svg_FOUND)
    target_link_libraries(cruciblecode PRIVATE Qt6::Svg)          # SVG images (also makes windeployqt ship the plugin)
    target_compile_definitions(cruciblecode PRIVATE CRUCIBLE_HAVE_SVG)
endif()
if(Qt6Pdf_FOUND AND Qt6PdfWidgets_FOUND)
    target_link_libraries(cruciblecode PRIVATE Qt6::Pdf Qt6::PdfWidgets)   # PDF viewer
    target_compile_definitions(cruciblecode PRIVATE CRUCIBLE_HAVE_PDF)
endif()

# live-preview server script (run with the system Python)
qt_add_resources(cruciblecode "app_web"
    PREFIX "/web"
    BASE "${CMAKE_CURRENT_SOURCE_DIR}/resources/web"
    FILES resources/web/live_server.py)

# built-in themes: resources/themes/*.css (generated by tools/make_themes.py, same format as the Theme Marketplace)
file(GLOB CRUCIBLE_THEME_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/resources/themes/*.css")
qt_add_resources(cruciblecode "app_themes"
    PREFIX "/themes"
    BASE "${CMAKE_CURRENT_SOURCE_DIR}/resources/themes"
    FILES ${CRUCIBLE_THEME_FILES})

# bundled English spelling dictionary (SCOWL word list, see resources/dict/COPYRIGHT-SCOWL.txt)
# stored uncompressed: the spell checker binary-searches it in place (file-backed, never copied to the heap)
qt_add_resources(cruciblecode "app_dict"
    PREFIX "/dict"
    BASE "${CMAKE_CURRENT_SOURCE_DIR}/resources/dict"
    OPTIONS -no-compress
    FILES resources/dict/en_US.txt)

# ---- logo / icons -----------------------------------------------------------
qt_add_resources(cruciblecode "app_icons"
    PREFIX "/icons"
    BASE "${CMAKE_CURRENT_SOURCE_DIR}/resources/icons"
    FILES
        resources/icons/icon-16.png  resources/icons/icon-24.png  resources/icons/icon-32.png
        resources/icons/icon-48.png  resources/icons/icon-64.png  resources/icons/icon-128.png
        resources/icons/icon-256.png resources/icons/icon-512.png)

if(WIN32)
    target_sources(cruciblecode PRIVATE resources/app.rc)       # exe icon in Explorer / taskbar
endif()
if(APPLE)
    set(CRUCIBLE_ICNS "${CMAKE_CURRENT_SOURCE_DIR}/resources/icons/icon.icns")
    target_sources(cruciblecode PRIVATE ${CRUCIBLE_ICNS})
    set_source_files_properties(${CRUCIBLE_ICNS} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(cruciblecode PROPERTIES MACOSX_BUNDLE_ICON_FILE icon.icns)
endif()
target_compile_definitions(cruciblecode PRIVATE CRUCIBLE_VERSION="${PROJECT_VERSION}")

# forkpty() lives in libutil on Linux (merged into libc since glibc 2.34)
if(UNIX AND NOT APPLE)
    find_library(CRUCIBLE_UTIL_LIB util)
    if(CRUCIBLE_UTIL_LIB)
        target_link_libraries(cruciblecode PRIVATE ${CRUCIBLE_UTIL_LIB})
    endif()
endif()

if(CRUCIBLE_HAVE_KSYNTAX)
    target_link_libraries(cruciblecode PRIVATE KF6::SyntaxHighlighting)
    target_compile_definitions(cruciblecode PRIVATE CRUCIBLE_HAVE_KSYNTAX)
endif()

if(MSVC)
    target_compile_options(cruciblecode PRIVATE /W4 /utf-8 /permissive-)
else()
    target_compile_options(cruciblecode PRIVATE -Wall -Wextra)
endif()

set_target_properties(cruciblecode PROPERTIES
    OUTPUT_NAME "CrucibleCode"
    MACOSX_BUNDLE_BUNDLE_NAME "CrucibleCode"
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.cruciblecode.app"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}")

# ---- install / deploy -------------------------------------------------------
install(TARGETS cruciblecode
    BUNDLE  DESTINATION .
    RUNTIME DESTINATION bin)

if(UNIX AND NOT APPLE)
    install(FILES resources/cruciblecode.desktop DESTINATION share/applications)
    foreach(sz 16 24 32 48 64 128 256 512)
        install(FILES resources/icons/icon-${sz}.png
                DESTINATION share/icons/hicolor/${sz}x${sz}/apps RENAME cruciblecode.png)
    endforeach()
endif()

if(Qt6_VERSION VERSION_GREATER_EQUAL 6.5)
    qt_generate_deploy_app_script(
        TARGET cruciblecode
        OUTPUT_SCRIPT cruciblecode_deploy_script
        NO_UNSUPPORTED_PLATFORM_ERROR)
    install(SCRIPT ${cruciblecode_deploy_script})
endif()
